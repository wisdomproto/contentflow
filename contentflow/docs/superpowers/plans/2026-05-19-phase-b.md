```javascript
// scripts/lib/extract-headings.mjs
export function extractHeadings(html) {
  if (!html) return { h1: '', h2: [] };
  const h1Match = html.match(/<h1[^>]*>([\s\S]*?)<\/h1>/i);
  const h1 = h1Match ? stripTags(h1Match[1]).trim() : '';

  const h2 = [];
  const h2Regex = /<h2[^>]*>([\s\S]*?)<\/h2>/gi;
  let match;
  while ((match = h2Regex.exec(html)) !== null) {
    h2.push(stripTags(match[1]).trim());
  }
  return { h1, h2 };
}

function stripTags(s) {
  return s.replace(/<[^>]+>/g, '').replace(/\s+/g, ' ');
}
```

```javascript
// scripts/lib/insert-images.mjs
const INJECTED_MARKER = 'phase-b-injected';
const INJECTED_REGEX = new RegExp(
  `<img[^>]*class="[^"]*${INJECTED_MARKER}[^"]*"[^>]*\\/?>\\s*`,
  'g'
);

export function insertImages(html, urls, alts) {
  // 1) 기존 injected img 모두 제거 (멱등)
  let cleaned = (html || '').replace(INJECTED_REGEX, '');

  const heroTag = urls.hero
    ? `<img src="${urls.hero}" alt="${esc(alts.h1)}" loading="lazy" class="${INJECTED_MARKER} blog-hero" />`
    : '';

  // 2) hero 삽입 — h1 직후, h1 없으면 맨 앞
  if (heroTag) {
    if (/<\/h1>/i.test(cleaned)) {
      cleaned = cleaned.replace(/(<\/h1>)/i, `$1${heroTag}`);
    } else {
      cleaned = heroTag + cleaned;
    }
  }

  // 3) h2 직후 section 삽입 (첫 두 번만)
  let h2Count = 0;
  cleaned = cleaned.replace(/<\/h2>/gi, (match) => {
    h2Count++;
    if (h2Count === 1 && urls.section1) {
      return `${match}<img src="${urls.section1}" alt="${esc(alts.h2_1)}" loading="lazy" class="${INJECTED_MARKER}" />`;
    }
    if (h2Count === 2 && urls.section2) {
      return `${match}<img src="${urls.section2}" alt="${esc(alts.h2_2)}" loading="lazy" class="${INJECTED_MARKER}" />`;
    }
    return match;
  });

  return cleaned;
}

function esc(s) {
  return String(s || '').replace(/[&<>"']/g, (c) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
  }[c]));
}
```

```javascript
// scripts/lib/comfyui-client.mjs
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, resolve } from 'node:path';
import WebSocket from 'ws';
import { COMFY_URL, COMFY_WS } from './phase-b-env.mjs';

const here = dirname(fileURLToPath(import.meta.url));
const TEMPLATE = JSON.parse(readFileSync(resolve(here, 'workflow-blog-illustration.json'), 'utf8'));

/**
 * Build workflow JSON with all placeholders substituted, using _meta.node_ids
 * to map abstract names to actual ComfyUI node IDs.
 * Pure function — exported separately for unit testing.
 */
export function buildWorkflowFromTemplate(template, opts) {
  const wf = JSON.parse(JSON.stringify(template));
  const ids = wf._meta?.node_ids;
  if (!ids) {
    throw new Error('workflow template missing _meta.node_ids — see Task 5 Step 4');
  }
  delete wf._meta;

  const seed = opts.seed ?? Math.floor(Math.random() * 1_000_000_000);
  if (wf[ids.ksampler_or_noise]) {
    wf[ids.ksampler_or_noise].inputs.seed = seed;
    // SamplerCustomAdvanced may use 'noise_seed' instead of 'seed'
    if ('noise_seed' in wf[ids.ksampler_or_noise].inputs) {
      wf[ids.ksampler_or_noise].inputs.noise_seed = seed;
    }
  }
  if (wf[ids.empty_latent]) {
    wf[ids.empty_latent].inputs.width = opts.width;
    wf[ids.empty_latent].inputs.height = opts.height;
  }
  if (wf[ids.positive_clip]) wf[ids.positive_clip].inputs.text = opts.positivePrompt;
  if (wf[ids.negative_clip]) wf[ids.negative_clip].inputs.text = opts.negativePrompt;
  if (wf[ids.save_image]) wf[ids.save_image].inputs.filename_prefix = `phase-b/${opts.contentId}-${opts.slot}`;

  return wf;
}

// Convenience wrapper using bundled template
export function buildWorkflow(opts) {
  return buildWorkflowFromTemplate(TEMPLATE, opts);
}

export async function checkComfyAlive() {
  try {
    const res = await fetch(`${COMFY_URL}/system_stats`, { signal: AbortSignal.timeout(3000) });
    return res.ok;
  } catch {
    return false;
  }
}

/**
 * Submit workflow and wait for completion. Returns PNG Buffer.
 * Timeout: 5 minutes per image.
 *
 * Race-condition fix: open WebSocket FIRST, await 'open', then submit prompt.
 * Otherwise ComfyUI may emit the completion event before the client subscribes.
 */
export async function generateImage(opts) {
  const clientId = `phase-b-${Math.random().toString(36).slice(2)}`;
  const wf = buildWorkflow(opts);

  // 1) Open WebSocket and wait for connection
  const ws = await openWs(clientId);

  let promptId = null;
  let settled = false;
  const TIMEOUT_MS = 5 * 60 * 1000;

  const completed = new Promise((resolveP, rejectP) => {
    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      try { ws.close(); } catch {}
      rejectP(new Error(`ComfyUI timeout (${TIMEOUT_MS / 1000}s)`));
    }, TIMEOUT_MS);

    ws.on('message', (raw) => {
      let msg;
      try { msg = JSON.parse(raw.toString()); } catch { return; }  // binary previews — ignore
      if (!promptId || msg.data?.prompt_id !== promptId) return;

      if (msg.type === 'executing' && msg.data?.node === null) {
        // Completed normally
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        try { ws.close(); } catch {}
        resolveP();
      } else if (msg.type === 'execution_error' || msg.type === 'execution_interrupted') {
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        try { ws.close(); } catch {}
        rejectP(new Error(`ComfyUI ${msg.type}: ${JSON.stringify(msg.data || {}).slice(0, 300)}`));
      }
    });
    ws.on('error', (err) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      rejectP(err);
    });
    ws.on('close', () => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      rejectP(new Error('ComfyUI WebSocket closed before completion'));
    });
  });

  // 2) Submit prompt
  const submitRes = await fetch(`${COMFY_URL}/prompt`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ prompt: wf, client_id: clientId }),
  });
  if (!submitRes.ok) {
    const body = await submitRes.text();
    try { ws.close(); } catch {}
    throw new Error(`ComfyUI /prompt failed: HTTP ${submitRes.status} ${body.slice(0, 200)}`);
  }
  const submitJson = await submitRes.json();
  promptId = submitJson.prompt_id;
  if (!promptId) {
    try { ws.close(); } catch {}
    throw new Error('ComfyUI /prompt returned no prompt_id');
  }

  // 3) Await completion (resolves via WebSocket handler above)
  await completed;

  // 4) Fetch history → output filename
  const histRes = await fetch(`${COMFY_URL}/history/${promptId}`);
  if (!histRes.ok) {
    throw new Error(`ComfyUI /history failed: HTTP ${histRes.status}`);
  }
  const hist = await histRes.json();
  const outputs = hist[promptId]?.outputs;
  if (!outputs) throw new Error(`ComfyUI history missing outputs for ${promptId}`);

  // 5) Find first image output
  for (const nodeId of Object.keys(outputs)) {
    const imgs = outputs[nodeId].images;
    if (imgs?.length) {
      const { filename, subfolder, type } = imgs[0];
      const viewUrl = `${COMFY_URL}/view?filename=${encodeURIComponent(filename)}&subfolder=${encodeURIComponent(subfolder || '')}&type=${type || 'output'}`;
      const pngRes = await fetch(viewUrl);
      if (!pngRes.ok) throw new Error(`ComfyUI /view failed: HTTP ${pngRes.status}`);
      return Buffer.from(await pngRes.arrayBuffer());
    }
  }
  throw new Error('No image found in ComfyUI outputs');
}

function openWs(clientId) {
  return new Promise((resolveP, rejectP) => {
    const ws = new WebSocket(`${COMFY_WS}?clientId=${clientId}`);
    const timer = setTimeout(() => rejectP(new Error('WS open timeout')), 10000);
    ws.once('open', () => { clearTimeout(timer); resolveP(ws); });
    ws.once('error', (err) => { clearTimeout(timer); rejectP(err); });
  });
}
```

```javascript
function validateMeta(meta) {
  const errors = [];

  if (!meta.url_slug || !/^[a-z0-9-]+$/.test(meta.url_slug) || meta.url_slug.length > 80) {

    errors.push(`meta_description length out of range: ${meta.meta_description?.length}`);

      if (attempt === 1) throw new Error(`validation failed twice: ${errs.join('; ')}`);
    } catch (err) {
      if (attempt === 1) throw err;

  return true;
}
```

```javascript
import { PutObjectCommand, HeadObjectCommand } from '@aws-sdk/client-s3';

  { name: 'section-1', width: 768, height: 768 },

  try { await r2.send(new HeadObjectCommand({ Bucket: R2_BUCKET, Key: key })); return true; }
  catch { return false; }
}

async function uploadWebp(key, pngBuffer) {
  const webp = await sharp(pngBuffer).webp({ quality: 85 }).toBuffer();
  await r2.send(new PutObjectCommand({
    Bucket: R2_BUCKET, Key: key, Body: webp, ContentType: 'image/webp',

  const { data: ba } = await supabase.from('base_articles')
    .select('body').eq('content_id', content.id).maybeSingle();

  // 어떤 슬롯이 필요한가?
  const needed = ['hero'];
  if (h2.length >= 1) needed.push('section-1');
  if (h2.length >= 2) needed.push('section-2');

  // 이미 R2에 있는 슬롯 skip
  const toGenerate = [];
  for (const slotName of needed) {
    const key = r2Key(content.id, slotName);
    if (await r2Exists(key)) {
      console.log(`  - ${slotName} exists, skip`);
    } else {
      toGenerate.push(slotName);
    }
  }
  if (toGenerate.length === 0) return true;

  // Prompts
  const prompts = await buildIllustrationPrompts({
    title: h1 || content.title, h2Texts: h2.slice(0, 2),
  });
  const promptMap = {
    'hero': prompts.hero,
    'section-1': prompts.section1,
    'section-2': prompts.section2,
  };

  // ComfyUI 호출 (1장씩 순차, 재시도 2회)
  for (const slotName of toGenerate) {
    const slot = SLOTS.find((s) => s.name === slotName);
    const positive = promptMap[slotName];
    if (!positive) { console.log(`  ✗ ${slotName} prompt missing`); continue; }

    let png;
    for (let attempt = 0; attempt < 2; attempt++) {
      try {
        const t0 = Date.now();
        png = await generateImage({
          positivePrompt: positive,
          negativePrompt: NEGATIVE_PROMPT,
          width: slot.width, height: slot.height,
          contentId: content.id, slot: slotName,
        });

      }

    if (!png) { console.log(`  ✗ ${slotName} failed after retries`); continue; }

  }

}
```

```javascript
    for (const c of all_contents) {
      const catKey = (c.category || '?').slice(0, 1);
      if (!seen[catKey]) { seen[catKey] = true; picked.push(c); }
    }
    return picked;
  }
  if (category) return all_contents.filter((c) => (c.category || '').startsWith(category));
  return all_contents;
}

async function processOne(content, logPath) {
  const t0 = Date.now();
  const ts = new Date().toISOString();
  logBoth(logPath, `${ts} [content=${content.id}] ${content.title.slice(0, 50)}`);

  const stages = [
    { name: 'seo', script: SCRIPTS.seo },
    { name: 'images', script: SCRIPTS.images },
    { name: 'insert', script: SCRIPTS.insert },
    { name: 'translate', script: SCRIPTS.translate },
  ];

  const results = {};
  for (const stage of stages) {
    const sT0 = Date.now();
    try {
      await runStage(stage.script, [`--content-id=${content.id}`]);
      results[stage.name] = 'OK';
      logBoth(logPath, `  ✓ ${stage.name} (${((Date.now() - sT0) / 1000).toFixed(1)}s)`);
    } catch (err) {
      results[stage.name] = `FAIL: ${err.message}`;
      logBoth(logPath, `  ✗ ${stage.name}: ${err.message}`);
      // 후속 단계는 계속 시도 (idempotent)
    }
  }
  logBoth(logPath, `  total ${((Date.now() - t0) / 1000).toFixed(1)}s | ${JSON.stringify(results)}`);
  return results;
}

async function main() {
  if (!flag('dry-run')) {
    if (!(await checkComfyAlive())) {
      console.error('✗ ComfyUI not running at http://127.0.0.1:8188');
      console.error('  Start: C:\\ComfyUI_windows_portable\\run_nvidia_gpu.bat');
      process.exit(1);
    }
  }

  mkdirSync(LOGS_DIR, { recursive: true });
  const ts = new Date().toISOString().replace(/[:.]/g, '-');
  const logPath = join(LOGS_DIR, `phase-b-${ts}.log`);
  writeFileSync(logPath, `# Phase B run ${ts}\n# args: ${ARGV.join(' ')}\n\n`, 'utf8');

  if (flag('retry-failed')) {
    // 최근 로그에서 FAIL 항목 content_id 추출 — 단순 구현: 가장 최근 로그
    const fs = await import('node:fs/promises');
    const files = (await fs.readdir(LOGS_DIR)).filter((f) => f.startsWith('phase-b-')).sort().reverse();
    if (files.length === 0) { console.log('No previous logs'); process.exit(0); }
    const recent = readFileSync(join(LOGS_DIR, files[0]), 'utf8');
    const failedIds = new Set();
    for (const line of recent.split('\n')) {
      const m = line.match(/\[content=([0-9a-f-]+)\].*FAIL/);
      if (m) failedIds.add(m[1]);
    }
    if (failedIds.size === 0) { console.log('No failed items in most recent log'); process.exit(0); }
    console.log(`Re-trying ${failedIds.size} failed contents from ${files[0]}`);
    const { data } = await supabase.from('contents').select('id, title, category, sort_order')
      .eq('project_id', PROJECT_ID).in('id', [...failedIds]);
    for (const c of data || []) await processOne(c, logPath);
    return;
  }

  const queue = await selectContents();
  logBoth(logPath, `Queue: ${queue.length} contents`);

  if (flag('dry-run')) {
    for (const c of queue) console.log(`  [DRY] ${c.id} ${c.title.slice(0, 60)}`);
    console.log(`\nDry run: ${queue.length} contents would be processed.`);
    return;
  }

  for (let i = 0; i < queue.length; i++) {
    const c = queue[i];
    logBoth(logPath, `\n[${i + 1}/${queue.length}]`);
    try { await processOne(c, logPath); }
    catch (err) { logBoth(logPath, `  ✗ orchestrator: ${err.message}`); }
  }
  logBoth(logPath, `\nDone. Log: ${logPath}`);
}

main().catch((e) => { console.error('Fatal:', e); process.exit(1); });
```
